# Una tarea lanzada desde el panel. Corre en segundo plano y avisa por burbuja
# cuando termina, leyendo el JSON que deja run-task.ps1 al salir.
function Invoke-OrbTask {
  param($Payload)

  if ($null -eq $Payload -or [string]::IsNullOrWhiteSpace($Payload.Prompt)) { return }

  $runner = Join-Path $PSScriptRoot '..\headless\run-task.ps1'
  if (-not (Test-Path $runner)) {
    Show-Bubble -Text (T 'task.noRunner') -Speaker 'GROK BOT' -Tone 'danger' -Anchor $window
    return
  }

  $done = Join-Path $env:TEMP ('dsh-orb-task-' + [guid]::NewGuid().ToString('N') + '.json')

  $runArgs = @('-NoProfile', '-ExecutionPolicy', 'Bypass', '-File', $runner, '-Task', $Payload.Prompt, '-OnDone', $done)
  if ($Payload.Path) { $runArgs += @('-WorkingDirectory', $Payload.Path) }
  if ($Payload.SessionId) { $runArgs += @('-SessionId', $Payload.SessionId) }

  if ($Payload.SessionId) {
    Show-Bubble -Text ((T 'task.resume') + ' ' + $Payload.SessionTitle) -Speaker 'GROK BOT' -Tone 'accent' -Anchor $window
  } else {
    Show-Bubble -Text ((T 'task.launched') + ' ' + $Payload.WorkspaceTitle) -Speaker 'GROK BOT' -Tone 'accent' -Anchor $window
  }

  # ProcessStartInfo con CreateNoWindow en vez de Start-Process -WindowStyle
  # Hidden: el segundo crea igualmente una consola que Windows puede llegar a
  # mostrar (fue exactamente el bug del orbe).
  try {
    $psi = New-Object System.Diagnostics.ProcessStartInfo
    $psi.FileName = 'powershell.exe'
    $quoted = foreach ($a in $runArgs) { if ($a -match '\s') { '"' + $a + '"' } else { $a } }
    $psi.Arguments = ($quoted -join ' ')
    $psi.UseShellExecute = $false
    $psi.CreateNoWindow = $true
    $psi.WindowStyle = [System.Diagnostics.ProcessWindowStyle]::Hidden
    [System.Diagnostics.Process]::Start($psi) | Out-Null
  } catch {
    Show-Bubble -Text (T 'task.failed') -Speaker 'GROK BOT' -Tone 'danger' -Anchor $window
    return
  }

  # Sondeo del archivo de resultado. La tarea puede tardar minutos; se abandona
  # a los 45 por si el runner murio sin escribir nada.
  $started = [DateTime]::Now
  $poll = New-Object System.Windows.Threading.DispatcherTimer
  $poll.Interval = [TimeSpan]::FromSeconds(4)
  $poll.Add_Tick({
      if (Test-Path -LiteralPath $done) {
        $poll.Stop()
        $res = $null
        try {
          $res = [System.IO.File]::ReadAllText($done, [System.Text.Encoding]::UTF8) | ConvertFrom-Json
          Remove-Item -LiteralPath $done -Force -ErrorAction SilentlyContinue
        } catch { }
        if ($null -eq $res) { return }
        if ($res.ok) {
          Show-Bubble -Text (T 'task.done') -Speaker 'GROK BOT' -Tone 'ok' -Anchor $window
        } else {
          $detail = [string]$res.tail
          if ($detail.Length -gt 110) { $detail = $detail.Substring(0, 107) + '...' }
          $text = T 'task.failed'
          if ($detail) { $text = $text + ' ' + $detail }
          Show-Bubble -Text $text -Speaker 'GROK BOT' -Tone 'danger' -Anchor $window
        }
        return
      }
      if (([DateTime]::Now - $started).TotalMinutes -gt 45) { $poll.Stop() }
    })
  $poll.Start()
}
